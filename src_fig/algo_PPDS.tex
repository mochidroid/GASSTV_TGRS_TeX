	   % \caption{The DP-PDS algorithm for \eqref{prob:denoising2PPDS}}
		\label{algo_DPPDS}
		\begin{algorithmic}[1]
			% \renewcommand{\algorithmicrequire}{\textbf{Input:}}
			% \renewcommand{\algorithmicensure}{\textbf{Output:}}
			\REQUIRE $\HSIClean^{(0)}, \NoiseSparse^{(0)}, \NoiseStripe^{(0)}, \VarDual{1}^{(0)}, \VarDual{2}^{(0)}, \VarDual{3}^{(0)}, \VarDual{4}^{(0)}$
			\ENSURE $\HSIClean^{(\IndexAlg)}$
			\WHILE {A stopping criterion is not satisfied}
				\STATE $\HSICleanTmp \leftarrow \DiffOpBandT \DiffOpSpT \VarDual{1}^{(\IndexAlg)} + \SpatialGraphIncidenceMat^{\top} \SpatialGraphWeightMat^{\top} \VarDual{2}^{(\IndexAlg)} + \VarDual{4}^{(\IndexAlg)}$
    			\STATE $\HSIClean^{(\IndexAlg+1)} \leftarrow \Projection{\SetRange}\left(\HSIClean^{(\IndexAlg)} - \ParamStepsize{\HSIClean} (\BalanceParam \DiffOpBandT \SpectralGraphLaplacian \DiffOpBand \HSIClean^{(\IndexAlg)} + \HSICleanTmp) \right)$;
    			\STATE $\NoiseSparse^{(\IndexAlg+1)} \leftarrow \Projection{\BallSparse} \left( \NoiseSparse^{(\IndexAlg)} - \ParamStepsize{\NoiseSparse} \VarDual{4}^{(\IndexAlg)} \right)$;
				\STATE $\NoiseStripe^{(\IndexAlg+1)} \leftarrow \Projection{\BallStripe} \left( \NoiseStripe^{(\IndexAlg)} - \ParamStepsize{\NoiseStripe} \bigl( \DiffOpVertT \VarDual{3}^{(\IndexAlg)} + \VarDual{4}^{(\IndexAlg)} \bigr) \right)$;
				\STATE $\ResHSIClean \leftarrow 2\HSIClean^{(\IndexAlg+1)} - \HSIClean^{(\IndexAlg)}$;
				\STATE $\ResNoiseSparse \leftarrow 2\NoiseSparse^{(\IndexAlg+1)} - \NoiseSparse^{(\IndexAlg)}$;
				\STATE $\ResNoiseStripe \leftarrow 2\NoiseStripe^{(\IndexAlg+1)} - \NoiseStripe^{(\IndexAlg)}$;
    			% \FOR{$\IndexBlock = 1, \ldots, \NumBlock$}
    			%     \STATE $\VarDualMatrix{1,\IndexBlock}^{'} \leftarrow \VarDualMatrix{1,\IndexBlock}^{(\IndexAlg)} + \ParamStepsize{\VarDualMatrix{1, \IndexBlock}} \ExpantionOp{\IndexBlock} \DiffOpSp \DiffOpBand \ResHSIClean$;
    			%     \STATE $\VarDualMatrix{1,\IndexBlock}^{(\IndexAlg+1)} \leftarrow \VarDualMatrix{1,\IndexBlock}^{'} - \ParamStepsize{\VarDualMatrix{1, \IndexBlock}} \prox_{\ParamStepsize{\VarDualMatrix{1, \IndexBlock}}^{-1}, \| \cdot \|_{*}} \left(\ParamStepsize{\VarDualMatrix{1, \IndexBlock}}^{-1} \VarDualMatrix{1,\IndexBlock}^{'}\right) $;
    		    % \ENDFOR
				\STATE $\VarDual{1}^{'} \leftarrow \VarDual{1}^{(\IndexAlg)} + \ParamStepsize{\VarDual{}} \DiffOpSp \DiffOpBand \ResHSIClean$;
    			\STATE $\VarDual{1}^{(\IndexAlg+1)} \leftarrow \VarDual{1}^{'} - \ParamStepsize{\VarDual{}} \prox_{\ParamStepsize{\VarDual{}} \| \cdot \|_{1}} \left(\frac{1}{\ParamStepsize{\VarDual{}}} \VarDual{1}^{'}  \right)$;
				\STATE $\VarDual{2}^{'} \leftarrow \VarDual{2}^{(\IndexAlg)} + \ParamStepsize{\VarDual{}} \SpatialGraphWeightMat \SpatialGraphIncidenceMat \ResHSIClean$;
    			\STATE $\VarDual{2}^{(\IndexAlg+1)} \leftarrow \VarDual{2}^{'} - \ParamStepsize{\VarDual{}} \Projection{\BallSpatialGraph} \left(\frac{1}{\ParamStepsize{\VarDual{}}} \VarDual{2}^{'}  \right)$;
				\STATE $\VarDual{3}^{(\IndexAlg+1)} \leftarrow \VarDual{3}^{(\IndexAlg)} + \ParamStepsize{\VarDual{}} \DiffOpVert \ResNoiseStripe$;
    			\STATE $\VarDual{4}^{'} \leftarrow \VarDual{4}^{(\IndexAlg)} + \ParamStepsize{\VarDual{}} \left( \ResHSIClean + \ResNoiseSparse + \ResNoiseStripe \right)$;
    			\STATE $\VarDual{4}^{(\IndexAlg+1)} \leftarrow \VarDual{4}^{'} - \ParamStepsize{\VarDual{}} \Projection{\BallFidel} \left(\frac{1}{\ParamStepsize{\VarDual{}}} \VarDual{4}^{'}  \right)$;
    			\STATE $\IndexAlg \leftarrow \IndexAlg + 1$;
			\ENDWHILE
		\end{algorithmic}
	\end{algorithm}
