	\begin{algorithm}[H]
	    \caption{P-PDS-based solver for (18)}
	   % \caption{The DP-PDS algorithm for \eqref{prob:denoising2PPDS}}
		\label{algo_DPPDS}
		\begin{algorithmic}[1]
			% \renewcommand{\algorithmicrequire}{\textbf{Input:}}
			% \renewcommand{\algorithmicensure}{\textbf{Output:}}
			\REQUIRE $\HSIClean^{(0)}, \NoiseSparse^{(0)}, \NoiseStripe^{(0)}, \VarDual{1}^{(0)}, \VarDual{2}^{(0)}, \VarDual{3}^{(0)}, \VarDual{4}^{(0)}$
			\ENSURE $\HSIClean^{(\IdxAlg)}$
			\WHILE {A stopping criterion is not satisfied}
				\STATE $\HSICleanTmp \leftarrow \DiffOpBandT \DiffOpSpT \VarDual{1}^{(\IdxAlg)} + \SpatialGraphIncidenceMat^{\top} \SpatialGraphWeightMat^{\top} \VarDual{2}^{(\IdxAlg)} + \VarDual{4}^{(\IdxAlg)}$
    			\STATE $\HSIClean^{(\IdxAlg+1)} \leftarrow \Projection{\SetRange}\left(\HSIClean^{(\IdxAlg)} - \ParamStepsize{\HSIClean} (\BalanceParam \DiffOpBandT \SpecGraphLaplacian \DiffOpBand \HSIClean^{(\IdxAlg)} + \HSICleanTmp) \right)$;
    			\STATE $\NoiseSparse^{(\IdxAlg+1)} \leftarrow \Projection{\BallSparse} \left( \NoiseSparse^{(\IdxAlg)} - \ParamStepsize{\NoiseSparse} \VarDual{4}^{(\IdxAlg)} \right)$;
				\STATE $\NoiseStripe^{(\IdxAlg+1)} \leftarrow \Projection{\BallStripe} \left( \NoiseStripe^{(\IdxAlg)} - \ParamStepsize{\NoiseStripe} \bigl( \DiffOpVertT \VarDual{3}^{(\IdxAlg)} + \VarDual{4}^{(\IdxAlg)} \bigr) \right)$;
				\STATE $\ResHSIClean \leftarrow 2\HSIClean^{(\IdxAlg+1)} - \HSIClean^{(\IdxAlg)}$;
				\STATE $\ResNoiseSparse \leftarrow 2\NoiseSparse^{(\IdxAlg+1)} - \NoiseSparse^{(\IdxAlg)}$;
				\STATE $\ResNoiseStripe \leftarrow 2\NoiseStripe^{(\IdxAlg+1)} - \NoiseStripe^{(\IdxAlg)}$;
    			% \FOR{$\IndexBlock = 1, \ldots, \NumBlock$}
    			%     \STATE $\VarDualMatrix{1,\IndexBlock}^{'} \leftarrow \VarDualMatrix{1,\IndexBlock}^{(\IdxAlg)} + \ParamStepsize{\VarDualMatrix{1, \IndexBlock}} \ExpantionOp{\IndexBlock} \DiffOpSp \DiffOpBand \ResHSIClean$;
    			%     \STATE $\VarDualMatrix{1,\IndexBlock}^{(\IdxAlg+1)} \leftarrow \VarDualMatrix{1,\IndexBlock}^{'} - \ParamStepsize{\VarDualMatrix{1, \IndexBlock}} \prox_{\ParamStepsize{\VarDualMatrix{1, \IndexBlock}}^{-1}, \| \cdot \|_{*}} \left(\ParamStepsize{\VarDualMatrix{1, \IndexBlock}}^{-1} \VarDualMatrix{1,\IndexBlock}^{'}\right) $;
    		    % \ENDFOR
				\STATE $\VarDual{1}^{'} \leftarrow \VarDual{1}^{(\IdxAlg)} + \ParamStepsize{\VarDual{}} \DiffOpSp \DiffOpBand \ResHSIClean$;
    			\STATE $\VarDual{1}^{(\IdxAlg+1)} \leftarrow \VarDual{1}^{'} - \ParamStepsize{\VarDual{}} \prox_{\ParamStepsize{\VarDual{}} \| \cdot \|_{1}} \left(\frac{1}{\ParamStepsize{\VarDual{}}} \VarDual{1}^{'}  \right)$;
				\STATE $\VarDual{2}^{'} \leftarrow \VarDual{2}^{(\IdxAlg)} + \ParamStepsize{\VarDual{}} \SpatialGraphWeightMat \SpatialGraphIncidenceMat \ResHSIClean$;
    			\STATE $\VarDual{2}^{(\IdxAlg+1)} \leftarrow \VarDual{2}^{'} - \ParamStepsize{\VarDual{}} \Projection{\BallSpatialGraph} \left(\frac{1}{\ParamStepsize{\VarDual{}}} \VarDual{2}^{'}  \right)$;
				\STATE $\VarDual{3}^{(\IdxAlg+1)} \leftarrow \VarDual{3}^{(\IdxAlg)} + \ParamStepsize{\VarDual{}} \DiffOpVert \ResNoiseStripe$;
    			\STATE $\VarDual{4}^{'} \leftarrow \VarDual{4}^{(\IdxAlg)} + \ParamStepsize{\VarDual{}} \left( \ResHSIClean + \ResNoiseSparse + \ResNoiseStripe \right)$;
    			\STATE $\VarDual{4}^{(\IdxAlg+1)} \leftarrow \VarDual{4}^{'} - \ParamStepsize{\VarDual{}} \Projection{\BallFidel} \left(\frac{1}{\ParamStepsize{\VarDual{}}} \VarDual{4}^{'}  \right)$;
    			\STATE $\IdxAlg \leftarrow \IdxAlg + 1$;
			\ENDWHILE
		\end{algorithmic}
	\end{algorithm}
